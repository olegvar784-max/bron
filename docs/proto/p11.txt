# 26. Контрольная запись до резервного копирования
$ psql -d bron -c "SELECT b.id, r.name, u.login, b.starts_at AT TIME ZONE 'Europe/Moscow' AS start_msk, b.ends_at AT TIME ZONE 'Europe/Moscow' AS end_msk, b.status FROM bookings b JOIN resources r ON r.id=b.resource_id JOIN users u ON u.id=b.user_id WHERE b.id = 20;"

# 27. Резервная копия (формат custom)
$ pg_dump -U postgres -h localhost -Fc -d bron -f backup/bron_20261102_1723.dump  &&  ls -lh backup/bron_20261102_1723.dump
21K backup/bron_20261102_1723.dump

# 28. Восстановление в отдельную базу bron_restore
$ createdb -U postgres -h localhost bron_restore
$ pg_restore -U postgres -h localhost -d bron_restore --no-owner backup/bron_20261102_1723.dump
pg_restore: завершено без ошибок
$ psql -d bron_restore -tAc "SELECT count(*) FROM bookings"
16

# 29. Запуск сервиса на восстановленной базе (DB_NAME=bron_restore, порт 8081) и проверка
$ DB_NAME=bron_restore PORT=8081 npm start   # в отдельном терминале
$ curl -s http://localhost:8081/health/ready
{"status":"ready"}
$ TOKEN=$(curl -s -X POST http://localhost:8081/api/auth/login -H 'Content-Type: application/json' -d '{"login":"ivanov","password":"…"}' | jq -r .token)
# вход выполнен, токен получен
$ curl -s http://localhost:8081/api/resources -H "Authorization: Bearer $TOKEN" | jq -c '.[] | [.id, .name]'

[7,"Ауд. 112"]
[2,"Ауд. 204"]

[5,"Лаборатория сетей"]
